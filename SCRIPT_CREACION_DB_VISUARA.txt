CREATE DATABASE Visuara;
GO
USE Visuara;
GO

CREATE TABLE dbo.Usuario (
    id_usuario        INT IDENTITY(1,1) NOT NULL,
    nombre            NVARCHAR(80) NOT NULL,
    apellido          NVARCHAR(80) NOT NULL,
    correo            NVARCHAR(100) NOT NULL,
    contrasena        VARCHAR(255) NOT NULL,
    tipo_discapacidad NVARCHAR(20) NULL,
    fecha_registro    DATETIME2(0) NOT NULL
        CONSTRAINT DF_Usuario_fecha DEFAULT (SYSDATETIME()),
    CONSTRAINT PK_Usuario PRIMARY KEY (id_usuario),
    CONSTRAINT UQ_Usuario_correo UNIQUE (correo),
    CONSTRAINT CK_Usuario_tipo_discapacidad
        CHECK (tipo_discapacidad IN (N'Congénita', N'Adquirida')),

    CONSTRAINT CK_Usuario_correo_formato
        CHECK (correo LIKE N'%_@_%._%' AND correo NOT LIKE N'% %')
);

CREATE TABLE dbo.Categoria (
    id_categoria     INT IDENTITY(1,1) NOT NULL,
    nombre_categoria NVARCHAR(80) NOT NULL,
    descripcion      NVARCHAR(300) NULL,
    CONSTRAINT PK_Categoria PRIMARY KEY (id_categoria),
    CONSTRAINT UQ_Categoria_nombre UNIQUE (nombre_categoria)
);

CREATE TABLE dbo.Dispositivo (
    id_dispositivo     INT IDENTITY(1,1) NOT NULL,
    nombre_dispositivo NVARCHAR(100) NOT NULL,
    tipo_dispositivo   NVARCHAR(50) NOT NULL,
    estado_conexion    NVARCHAR(20) NOT NULL
        CONSTRAINT DF_Dispositivo_estado DEFAULT (N'Desconectado'),
    CONSTRAINT PK_Dispositivo PRIMARY KEY (id_dispositivo),
    CONSTRAINT CK_Dispositivo_tipo
        CHECK (tipo_dispositivo IN (N'Celular', N'Lentes inteligentes', N'Cámara externa')),
    CONSTRAINT CK_Dispositivo_estado
        CHECK (estado_conexion IN (N'Conectado', N'Desconectado'))
);

CREATE TABLE dbo.Configuracion (
    id_configuracion  INT IDENTITY(1,1) NOT NULL,
    id_usuario        INT NOT NULL,
    velocidad_voz_tts DECIMAL(4,2) NOT NULL
        CONSTRAINT DF_Configuracion_velocidad DEFAULT (1.00),
    tono_voz          NVARCHAR(50) NULL,
    volumen_audio     INT NOT NULL
        CONSTRAINT DF_Configuracion_volumen DEFAULT (70),
    modo_deteccion    NVARCHAR(20) NOT NULL
        CONSTRAINT DF_Configuracion_modo DEFAULT (N'Manual'),
    CONSTRAINT PK_Configuracion PRIMARY KEY (id_configuracion),
    CONSTRAINT UQ_Configuracion_usuario UNIQUE (id_usuario),
    CONSTRAINT FK_Configuracion_Usuario
        FOREIGN KEY (id_usuario) REFERENCES dbo.Usuario(id_usuario),
    CONSTRAINT CK_Configuracion_velocidad
        CHECK (velocidad_voz_tts BETWEEN 0.50 AND 2.00),
    CONSTRAINT CK_Configuracion_volumen
        CHECK (volumen_audio BETWEEN 0 AND 100),
    CONSTRAINT CK_Configuracion_modo
        CHECK (modo_deteccion IN (N'Automático', N'Manual'))
);

CREATE TABLE dbo.Ambiente (
    id_ambiente     INT IDENTITY(1,1) NOT NULL,
    id_usuario      INT NOT NULL,
    nombre_ambiente NVARCHAR(80) NOT NULL,
    numero_piso     INT NULL,
    CONSTRAINT PK_Ambiente PRIMARY KEY (id_ambiente),
    CONSTRAINT FK_Ambiente_Usuario
        FOREIGN KEY (id_usuario) REFERENCES dbo.Usuario(id_usuario),

    CONSTRAINT UQ_Ambiente_ambiente_usuario UNIQUE (id_ambiente, id_usuario)
);

CREATE TABLE dbo.Objeto (
    id_objeto          INT IDENTITY(1,1) NOT NULL,
    id_categoria       INT NOT NULL,
    nombre_objeto      NVARCHAR(100) NOT NULL,
    descripcion_fisica NVARCHAR(500) NULL,
    ubicacion_habitual NVARCHAR(150) NULL,
    CONSTRAINT PK_Objeto PRIMARY KEY (id_objeto),
    CONSTRAINT FK_Objeto_Categoria
        FOREIGN KEY (id_categoria) REFERENCES dbo.Categoria(id_categoria)
);

CREATE TABLE dbo.UsuarioDispositivo (
    id_usuario         INT NOT NULL,
    id_dispositivo     INT NOT NULL,
    fecha_vinculacion  DATETIME2(0) NOT NULL
        CONSTRAINT DF_UsuarioDispositivo_fecha DEFAULT (SYSDATETIME()),
    estado_vinculacion NVARCHAR(20) NOT NULL,
    CONSTRAINT PK_UsuarioDispositivo PRIMARY KEY (id_usuario, id_dispositivo),
    CONSTRAINT FK_UsuarioDispositivo_Usuario
        FOREIGN KEY (id_usuario) REFERENCES dbo.Usuario(id_usuario),
    CONSTRAINT FK_UsuarioDispositivo_Dispositivo
        FOREIGN KEY (id_dispositivo) REFERENCES dbo.Dispositivo(id_dispositivo),
    CONSTRAINT CK_UsuarioDispositivo_estado
        CHECK (estado_vinculacion IN (N'Activo', N'Inactivo'))
);

CREATE TABLE dbo.UsuarioObjeto (
    id_usuario       INT NOT NULL,
    id_objeto        INT NOT NULL,
    fecha_asociacion DATETIME2(0) NOT NULL
        CONSTRAINT DF_UsuarioObjeto_fecha DEFAULT (SYSDATETIME()),
    tipo_vinculo     NVARCHAR(20) NOT NULL,
    CONSTRAINT PK_UsuarioObjeto PRIMARY KEY (id_usuario, id_objeto),
    CONSTRAINT FK_UsuarioObjeto_Usuario
        FOREIGN KEY (id_usuario) REFERENCES dbo.Usuario(id_usuario),
    CONSTRAINT FK_UsuarioObjeto_Objeto
        FOREIGN KEY (id_objeto) REFERENCES dbo.Objeto(id_objeto),
    CONSTRAINT CK_UsuarioObjeto_tipo
        CHECK (tipo_vinculo IN (N'Propio', N'Compartido'))
);

CREATE TABLE dbo.ImagenReferencia (
    id_imagen_ref INT IDENTITY(1,1) NOT NULL,
    id_objeto     INT NOT NULL,
    ruta_archivo  NVARCHAR(1000) NOT NULL,
    fecha_captura DATETIME2(0) NOT NULL
        CONSTRAINT DF_ImagenReferencia_fecha DEFAULT (SYSDATETIME()),
    CONSTRAINT PK_ImagenReferencia PRIMARY KEY (id_imagen_ref),
    CONSTRAINT FK_ImagenReferencia_Objeto
        FOREIGN KEY (id_objeto) REFERENCES dbo.Objeto(id_objeto)
);

CREATE TABLE dbo.Escaneo (
    id_escaneo        INT IDENTITY(1,1) NOT NULL,
    id_usuario        INT NOT NULL,
    id_dispositivo    INT NOT NULL,
    id_ambiente       INT NOT NULL,
    fecha_hora        DATETIME2(0) NOT NULL
        CONSTRAINT DF_Escaneo_fecha DEFAULT (SYSDATETIME()),
    nivel_iluminacion NVARCHAR(30) NULL,
    CONSTRAINT PK_Escaneo PRIMARY KEY (id_escaneo),
    CONSTRAINT FK_Escaneo_Usuario
        FOREIGN KEY (id_usuario) REFERENCES dbo.Usuario(id_usuario),
    CONSTRAINT FK_Escaneo_Dispositivo
        FOREIGN KEY (id_dispositivo) REFERENCES dbo.Dispositivo(id_dispositivo),
    CONSTRAINT FK_Escaneo_Ambiente
        FOREIGN KEY (id_ambiente) REFERENCES dbo.Ambiente(id_ambiente),
    CONSTRAINT FK_Escaneo_AmbienteUsuario
        FOREIGN KEY (id_ambiente, id_usuario)
        REFERENCES dbo.Ambiente(id_ambiente, id_usuario),
    CONSTRAINT FK_Escaneo_UsuarioDispositivo
        FOREIGN KEY (id_usuario, id_dispositivo)
        REFERENCES dbo.UsuarioDispositivo(id_usuario, id_dispositivo),
    CONSTRAINT CK_Escaneo_iluminacion
        CHECK (nivel_iluminacion IN (N'Alta', N'Media', N'Baja'))
);

CREATE TABLE dbo.Deteccion (
    id_deteccion       INT IDENTITY(1,1) NOT NULL,
    id_escaneo         INT NOT NULL,
    id_objeto          INT NULL,
    porcentaje_certeza DECIMAL(5,2) NOT NULL,
    mensaje_audio      NVARCHAR(1000) NULL,
    CONSTRAINT PK_Deteccion PRIMARY KEY (id_deteccion),
    CONSTRAINT FK_Deteccion_Escaneo
        FOREIGN KEY (id_escaneo) REFERENCES dbo.Escaneo(id_escaneo),
    CONSTRAINT FK_Deteccion_Objeto
        FOREIGN KEY (id_objeto) REFERENCES dbo.Objeto(id_objeto),
    CONSTRAINT CK_Deteccion_certeza
        CHECK (porcentaje_certeza BETWEEN 0 AND 100),

    CONSTRAINT UQ_Deteccion_deteccion_objeto UNIQUE (id_deteccion, id_objeto)
);

CREATE TABLE dbo.HistorialUbicacion (
    id_historial   INT IDENTITY(1,1) NOT NULL,
    id_objeto      INT NOT NULL,
    id_deteccion   INT NOT NULL,
    fecha_registro DATETIME2(0) NOT NULL
        CONSTRAINT DF_HistorialUbicacion_fecha DEFAULT (SYSDATETIME()),
    CONSTRAINT PK_HistorialUbicacion PRIMARY KEY (id_historial),
    CONSTRAINT UQ_HistorialUbicacion_deteccion UNIQUE (id_deteccion),
    CONSTRAINT FK_HistorialUbicacion_Objeto
        FOREIGN KEY (id_objeto) REFERENCES dbo.Objeto(id_objeto),
    CONSTRAINT FK_HistorialUbicacion_Deteccion
        FOREIGN KEY (id_deteccion) REFERENCES dbo.Deteccion(id_deteccion),

    CONSTRAINT FK_HistorialUbicacion_DeteccionObjeto
        FOREIGN KEY (id_deteccion, id_objeto)
        REFERENCES dbo.Deteccion(id_deteccion, id_objeto)
);
