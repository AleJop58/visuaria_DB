
const visuaraDB = db.getSiblingDB("Visuara");

if (visuaraDB.getCollectionNames().some(nombre => ["Objetos", "Escaneos"].includes(nombre))) {
    throw new Error("Las colecciones Objetos o Escaneos ya existen. Ejecuta este archivo sobre una base de datos sin esas colecciones.");
}

const esquemaObjetos = {
    bsonType: "object",
    title: "Objetos",
    additionalProperties: false,
    required: ["id_objeto", "id_usuario", "id_categoria", "nombre_objeto", "imagenes_referencia"],
    properties: {
        _id: { bsonType: "objectId" },
        id_objeto: { bsonType: "objectId" },
        id_usuario: { bsonType: "objectId" },
        id_categoria: { bsonType: "objectId" },
        nombre_objeto: { bsonType: "string", minLength: 1, maxLength: 100 },
        descripcion_fisica: { bsonType: "string", maxLength: 500 },
        ubicacion_habitual: { bsonType: "string", maxLength: 150 },
        imagenes_referencia: {
            bsonType: "array",
            items: {
                bsonType: "object",
                additionalProperties: false,
                required: ["id_imagen_ref", "ruta_archivo", "fecha_captura"],
                properties: {
                    id_imagen_ref: { bsonType: "objectId" },
                    ruta_archivo: { bsonType: "string", minLength: 1, maxLength: 1000 },
                    fecha_captura: { bsonType: "date" }
                }
            }
        }
    }
};

const esquemaEscaneos = {
    bsonType: "object",
    title: "Escaneos",
    additionalProperties: false,
    required: ["id_escaneo", "id_usuario", "id_dispositivo", "id_ambiente", "fecha_hora", "detecciones"],
    properties: {
        _id: { bsonType: "objectId" },
        id_escaneo: { bsonType: "objectId" },
        id_usuario: { bsonType: "objectId" },
        id_dispositivo: { bsonType: "objectId" },
        id_ambiente: { bsonType: "objectId" },
        fecha_hora: { bsonType: "date" },
        nivel_iluminacion: { bsonType: "string", enum: ["Alta", "Media", "Baja"] },
        detecciones: {
            bsonType: "array",
            items: {
                bsonType: "object",
                additionalProperties: false,
                required: ["id_deteccion", "id_objeto", "porcentaje_certeza", "mensaje_audio"],
                properties: {
                    id_deteccion: { bsonType: "objectId" },
                    id_objeto: { bsonType: ["objectId", "null"] },
                    porcentaje_certeza: { bsonType: "number", minimum: 0, maximum: 100 },
                    mensaje_audio: { bsonType: "string", maxLength: 1000 }
                }
            }
        }
    }
};

visuaraDB.createCollection("Objetos", {
    capped: false,
    validator: { $jsonSchema: esquemaObjetos },
    validationLevel: "strict",
    validationAction: "error"
});

visuaraDB.createCollection("Escaneos", {
    capped: false,
    validator: { $jsonSchema: esquemaEscaneos },
    validationLevel: "strict",
    validationAction: "error"
});

visuaraDB.Objetos.createIndex({ id_objeto: 1 }, { unique: true });
visuaraDB.Objetos.createIndex({ id_usuario: 1, nombre_objeto: 1 });
visuaraDB.Escaneos.createIndex({ id_escaneo: 1 }, { unique: true });
visuaraDB.Escaneos.createIndex({ id_usuario: 1, fecha_hora: -1 });
visuaraDB.Escaneos.createIndex({ "detecciones.id_objeto": 1, fecha_hora: -1 });
